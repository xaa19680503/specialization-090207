function xaa_img.speed_fire(m:array of timage):string;
var h,w,i,n:integer; s:string; //  локальные переменные определили
begin  n:=length(m)-1;   h:=-1;w:=-1; //   начальные данные
  if hana>=0 then  //энергии более нуля
  for i:=0 to n do      begin//  цикл переборки объектов пуль
      if m[i]=nil then continue;  // пулю потратили
      if sizeof(m[i]) = 0 then continue;  //  доп. контроль, что пули нет
      if m[i].ShowHint then continue;   // дот контроль  что пуля пропала  с видимости
      h:=m[i].top;       w:=m[i].left;  //  уточняем  координаты  пули
      if h>top then                    // пуля  ниже верха
      if h<top+height then    //  пуля выше   ступни
      if w>left then                //  пуля  за  гранью
      if w<left+width then  //  пуля  не вылетела из граней
         begin dec(hana);    //  пуля  попала и  снижение уровня жизни
         m[i].Hide; //m[i].free; m[i]:=nil; //  стираем  пулю  убираем объект  пули
         if hana<0 then hide;           //  если  жизни хана  то убираем цель
         end;
      end; s:=inttostr(hana)+#13+#10+  //  отображаем данные
              inttostr(top)+#13+#10+  //  перевод каретки =+#13+#10
              inttostr(left)+#13+#10+
              inttostr(top+height)+#13+#10+
              inttostr(width+left)+#13+#10+
              inttostr(h)+#13+#10+ inttostr(w);
   result:='live = '+s;  //  данные на результат передаём
end;



   function xaa_img.create_fire(Sender: TObject):timage;
   var LN:integer;         //  локальную переменную определили
   begin   LN:=length(img);    setlength(img,ln+1);  //  добавим пулю в очедь
       img[ln]:=timage.Create(self);           //  создадим объект  под  пулю в очереди
       img[ln].Canvas.Pen.Color:=clred;  //  укажем цвет пули
       img[ln].Parent:=parent;                 //   укажем где будет пуля отображаться
       img[ln].Top:=top+5;    img[ln].Left:=width+left+5;  //  укажем координаты пули начальные
       img[ln].width:=15;    img[ln].Height:=5;                    //   укажем размер пули
       img[ln].Canvas.Rectangle(1,1,img[ln].width-1,img[ln].height-1); //  рисуем пулю
       img[ln].Show;      //  покажем пулю
      result:=img[ln];  //  пулю передали на результат
   end;

function xaa_img.key_xy(k:char):string;
var s:string; //  локальную переменную определили
begin          // обеспечим характер движения
if  FOffsety=0 then FOffsety:=5;
if k=chr_fr_r  then begin spd_x:=0;    create_fire(self); s:=k; end else //выстрел в право
if k=chr_fr_L  then begin spd_x:=0;    create_fire_LEFT(self); s:=k; end else //выстрел в право
if k=chr_fr_d  then begin spd_y:=0;    create_fire_down(self); s:=k; end else //выстрел в право
if k=chr_fr_u  then begin spd_y:=0;    create_fire_up(self); s:=k; end else //выстрел в право
if k=chr_up    then begin spd_x:=0;    spd_y:=-FOffsety;  s:=k; end else
if k=chr_down  then begin spd_x:=0;    spd_y:=FOffsety;   s:=k; end else
if k=chr_left  then begin spd_x:=-FOffsety; spd_y:=0;     s:=k; end else
if k=chr_right then begin spd_x:= FOffsety; spd_y:=0;     s:=k; end else
if k=chr_stop  then begin spd_x:=0;         spd_y:=0;     s:=k; end;

  result:=s+' '+inttostr(FOffsetX)+'  ' +inttostr( OffsetX)+'  ' +inttostr( OffsetY)+' '+k;
end;
function xaa_img.go_xy(wdth,hght:integer):string;
      var s:string;      n,i:integer;
begin   Left := Max(0, Min(Wdth - Width, Left +spd_x));  // обеспечим движение     -----------
                   Top := Max(0, Min(Hght - Height, Top + spd_y));


        n:=-1;   // начальное значение  наличия  пуль в трассе СЛЕВА
    for i:=0  to length(umg)-1 do
          begin  if umg[i] = nil then continue; // контроль наличия пули
               if sizeof(umg[i]) = 0 then continue;  // повторный контроль
               if umg[i].top>0        then umg[i].top:=umg[i].top-3; // перемещение пули
               if umg[i].top<5  then begin umg[i].Hide;umg[i].free; umg[i]:=nil;
                          continue ;  end;  //улетелаs
               inc(n); //если пуля есть, то её считаем
           end; if n<0 then setlength(umg,0) ;    //если пуль нет, то  стираем  трассу пуль

            n:=-1;   // начальное значение  наличия  пуль в трассе СЛЕВА
       for i:=0  to length(Lmg)-1 do
           begin  if Lmg[i] = nil then continue; // контроль наличия пули
                  if sizeof(Lmg[i]) = 0 then continue;  // повторный контроль
                  if lmg[i].left>0    then Lmg[i].Left:=Lmg[i].Left-3; // перемещение пули
                  if Lmg[i].left<5  then begin    Lmg[i].Hide;  Lmg[i].free; Lmg[i]:=nil;
                                                    continue ;  end;  //улетела
                   inc(n); //если пуля есть, то её считаем
           end; if n<0 then setlength(Lmg,0) ;    //если пуль нет, то  стираем  трассу пуль

           n:=-1;   // начальное значение  наличия  пуль в трассе СПРАВА
        for i:=0  to length(img)-1 do
            begin                if img[i] = nil then continue; // контроль наличия пули
                          if sizeof(img[i]) = 0 then continue;  // повторный контроль
                 if img[i].left<400+left    then img[i].Left:=img[i].Left+3; // перемещение пули
                if img[i].left>390 +left  then begin    img[i].Hide;   img[i].free;    img[i]:=nil;
                                                continue ;  end;  //улетела
               inc(n); //если пуля есть, то её считаем
            end;      if n<0 then setlength(img,0) ;    //если пуль нет, то  стираем  трассу пуль


            n:=-1;   // начальное значение  наличия  пуль в трассе  вниз
        for i:=0  to length(dmg)-1 do
            begin                if dmg[i] = nil then continue; // контроль наличия пули
                          if sizeof(dmg[i]) = 0 then continue;  // повторный контроль
                if dmg[i].top<400+height  then dmg[i].top:=dmg[i].top+3; // перемещение пули
                if dmg[i].top>390 +height then begin    dmg[i].Hide;   dmg[i].free;    dmg[i]:=nil;
                          continue ;  end;  //улетела
               inc(n); //если пуля есть, то её считаем
            end;      if n<0 then setlength(dmg,0) ;    //если пуль нет, то  стираем  трассу пуль

                     s:='t '+inttostr(top)+' L '+inttostr(left)+ ' y '+inttostr(spd_y)+   ' x '+inttostr(spd_x);
            result:=s+' n = '+inttostr(n); //отображаем  результат
end;

function  xaa_img.DestRect: TRect;
begin    Result := inherited DestRect();  Transparent:=true;
 if (FOffsetX <> 0) or (FOffsetY <> 0) then
 LCLProc.OffsetRect(Result, -FOffsetX, -FOffsetY); end;


procedure  xaa_img.blink(limit,predel:integer);
begin    OffsetX := (OffsetX + limit) mod predel;
end;

procedure  xaa_img.SetOffsetX(AValue: Integer); // опишем процедуру
begin   if FOffsetX = AValue then exit;   FOffsetX := AValue;  PictureChanged(Self);
 end;
procedure  xaa_img.SetOffsetY(AValue: Integer); // опишем  процедуру
begin    if FOffsetY = AValue then exit;    FOffsetY := AValue;  PictureChanged(Self);
end;
